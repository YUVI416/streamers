 private void GlobalHook_KeyUp(object sender, KeyEventArgs e)
 {
     if (isListeningForKey)
     {
         toggleKey = e.KeyCode;
         toggleMouseButton = MouseButtons.None;
         toggleWheelDirection = 0;


         isListeningForKey = false;
     }
     else if (e.KeyCode == toggleKey)
     {
         ToggleSniperScope();
     }
 }


 private void GlobalHook_MouseUp(object sender, MouseEventArgs e)
 {
     if (isListeningForKey)
     {
         toggleMouseButton = e.Button;
         toggleKey = Keys.None;
         toggleWheelDirection = 0;


         isListeningForKey = false;
     }
     else
     {
         if (e.Button == toggleMouseButton || e.Button == MouseButtons.Left)
         {
             ToggleSniperScope();
         }
     }
 }



 private void GlobalHook_MouseWheel(object sender, MouseEventArgs e)
 {
     if (isListeningForKey)
     {
         toggleWheelDirection = e.Delta > 0 ? 1 : -1;
         toggleKey = Keys.None;
         toggleMouseButton = MouseButtons.None;
         string direction = toggleWheelDirection == 1 ? "Wheel UP" : "Wheel DOWN";

         status.Text = $"Sniper Toggle set to {direction}";
         isListeningForKey = false;
     }
     else
     {
         if ((toggleWheelDirection == 1 && e.Delta > 0) ||
             (toggleWheelDirection == -1 && e.Delta < 0))
         {
             ToggleSniperScope();
         }
     }
 }

 private DateTime lastToggleAttempt = DateTime.MinValue;
 private readonly int toggleCooldownMs = 200;

 private async void ToggleSniperScope()
 {
     var now = DateTime.Now;
     if ((now - lastToggleAttempt).TotalMilliseconds < toggleCooldownMs)
         return;

     lastToggleAttempt = now;

     if (!isSniperScopeOn)
     {

         sniperscopeoff();

     }


     isSniperScopeOn = !isSniperScopeOn;
     lastToggleTime = now;
 }
